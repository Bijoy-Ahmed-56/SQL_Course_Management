create database student;

use student;

CREATE TABLE Student (
	Sno char (10) PRIMARY KEY  ,
	Sname char (20)  NOT NULL ,
	Ssex char (10)  ,
	Sage tinyint  ,
	Sdept char (20) 
);

CREATE TABLE Course (
	Cno char (10) PRIMARY KEY ,
	Cname char (20)  NOT NULL ,
	Ccredit tinyint  ,
	Semester tinyint  
);

CREATE TABLE sc (
	sno char (10) NOT NULL ,
	cno char (10) NOT NULL ,
	mark smallint  ,
	c_type char (10)  ,
	PRIMARY KEY(sno,cno),
	CHECK (mark >= 0 and mark <= 100)
);

select Sname , Sno , year(curdate()) - Sage AS birth_year
from Student ;

select count(Sno) as total_number_student
from Student ;

SELECT 
    cno AS course_id, 
    AVG(mark) AS average_score, 
    COUNT(sno) AS number_of_students
FROM sc
GROUP BY cno;


SELECT sno AS student_id
FROM sc
GROUP BY sno
HAVING AVG(mark) > 80;


SELECT 
    Student.Sno AS student_id, 
    Student.Sname AS student_name, 
    sc.cno AS course_id, 
    sc.mark AS score
FROM Student
LEFT JOIN sc ON Student.Sno = sc.sno;


SELECT 
    Student.Sname AS student_name, 
    sc.cno AS course_id, 
    sc.mark AS score
FROM Student
JOIN sc ON Student.Sno = sc.sno
WHERE Student.Sdept = 'CS';


SELECT 
    Student.Sname AS student_name, 
    Course.Cname AS course_name, 
    sc.mark AS score
FROM Student
JOIN sc ON Student.Sno = sc.sno
JOIN Course ON sc.cno = Course.Cno
WHERE Student.Sdept = 'IS' AND Course.Cname = 'VB';

SELECT 
    Student.Sname AS student_name, 
    Student.Sdept AS department
FROM Student
JOIN sc ON Student.Sno = sc.sno
JOIN Course ON sc.cno = Course.Cno
WHERE Course.Cname = 'VB';
